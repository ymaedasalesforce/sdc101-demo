function sendMockData() { let baseUrl = document.getElementById('endpoint').value.trim(); const appSourceId = document.getElementById('appSourceId').value.trim(); const payloadInput = document.getElementById('payload').value.trim(); const statusDiv = document.getElementById('status'); statusDiv.style.display = "none"; statusDiv.className = ""; if (!baseUrl || !appSourceId || !payloadInput) { showStatus("すべての項目を入力してください。", "error"); return; } // ★修正ポイント: https:// から始まっていない場合は自動で付与する if (!baseUrl.startsWith('http://') && !baseUrl.startsWith('https://')) { baseUrl = 'https://' + baseUrl; } if (!baseUrl.endsWith('/v1/event')) { baseUrl = baseUrl.replace(/\/$/, '') + '/v1/event'; } let userPayload; try { userPayload = JSON.parse(payloadInput); } catch (e) { showStatus("JSONの形式が正しくありません。構文を確認してください。", "error"); return; } const finalPayload = { "appSourceId": appSourceId, "eventTime": new Date().toISOString(), ...userPayload }; fetch(baseUrl, { method: "POST", headers: { "Content-Type": "application/json" }, body: JSON.stringify(finalPayload) }) .then(response => { if (response.ok) { showStatus(`✅ 送信成功！ ステータス: ${response.status} (${response.statusText})`, "success"); } else { showStatus(`❌ 送信失敗。サーバーがエラーを返しました。 ステータス: ${response.status}`, "error"); } }) .catch(error => { console.error("通信エラー詳細:", error); showStatus("❌ 通信エラー。URLが間違っているか、CORS制限が発生している可能性があります。ブラウザのコンソールログを確認してください。", "error"); }); }
